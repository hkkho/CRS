### Report structure and final scope - T0

```powershell
$reports = @('docs/gates/G0-R1.md', 'docs/tasks/G0-R1.md')
foreach ($report in $reports) {
    $content = Get-Content -Raw -LiteralPath $report
    if (([regex]::Matches($content, '(?m)^```').Count % 2) -ne 0) {
        throw "$report has unbalanced code fences."
    }
    if (Get-Content -LiteralPath $report |
        Where-Object { $_ -match '[ \t]+$' }) {
        throw "$report contains trailing whitespace."
    }
    $base = Split-Path -Parent (Resolve-Path -LiteralPath $report)
    foreach ($match in [regex]::Matches(
            $content, '\[[^\]]+\]\((?<target>[^)\r\n]+)\)')) {
        $target = $match.Groups['target'].Value
        if (-not [Uri]::IsWellFormedUriString(
                $target, [UriKind]::Absolute)) {
            $relative = $target.Split('#')[0]
            if ($relative -and
                -not (Test-Path -LiteralPath (Join-Path $base $relative))) {
                throw "Missing local link: $target"
            }
        }
    }
}
