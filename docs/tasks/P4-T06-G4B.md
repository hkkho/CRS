~~~powershell
$ErrorActionPreference = 'Stop'
$files = @(
  'tests/ReactorSim.Golden.Tests/P4T06G4BAdmittedCandidateConsumerTests.cs',
  'tests/ReactorSim.Golden.Tests/ReactorSim.Golden.Tests.csproj',
  'data/comparisons/p4-t06-r5-candidate-snapshots-v1.json',

  'benchmarks/P4-T08-static-solver-benchmark.json'
)
$slash = [char]92
$forward = [char]47
$hostMarkers = @(
  'C:' + $slash + 'Users' + $slash,
  $forward + 'Users' + $forward,
  $forward + 'home' + $forward
)
foreach ($file in $files) {
  if (-not (Test-Path -LiteralPath $file -PathType Leaf)) { throw "Missing $file" }
  $text = [IO.File]::ReadAllText((Join-Path (Get-Location) $file))
  if ($text -match '(?m)[ \t]+$') { throw "Trailing whitespace: $file" }
  if ($text.IndexOf([char]0xFFFD) -ge 0) { throw "Replacement character: $file" }
  foreach ($character in $text.ToCharArray()) {
    if ([int][char]$character -gt 127) { throw "Non-ASCII character: $file" }
  }
  foreach ($marker in $hostMarkers) {
    if ($text.IndexOf($marker, [StringComparison]::OrdinalIgnoreCase) -ge 0) { throw "Host path: $file" }
  }
}
$outputPath = Join-Path (Get-Location) 'data/comparisons/p4-t06-r5-candidate-snapshots-v1.json'
$output = Get-Content -LiteralPath $outputPath -Raw | ConvertFrom-Json
if ($output.status -ne 'candidate' -or $output.evidence_class -ne 'synthetic' -or $output.evidence_approval -ne 'Candidate') { throw 'Candidate evidence status changed' }
$sha = (Get-FileHash -Algorithm SHA256 -LiteralPath $outputPath).Hash.ToLowerInvariant()
~~~
