def between(start, end):
    i = s.index(start)
    return s[i:s.index(end, i)]

def rows(section, header, width):
    lines, result = section.splitlines(), []
    for i, line in enumerate(lines):
        if line.strip() != header:
            continue
        assert i + 1 < len(lines) and lines[i + 1].strip().startswith('| ---'), 'bad table separator'
        j = i + 2
        while j < len(lines) and lines[j].strip().startswith('|'):
            raw = lines[j].strip()
            assert raw.endswith('|'), f'malformed table row: {raw}'
            cells = [x.strip() for x in raw[1:-1].split('|')]
            assert len(cells) == width and all(cells), f'malformed {width}-column row: {raw}'
            result.append(cells)
            j += 1
    assert result, f'table not found: {header}'
    return result
