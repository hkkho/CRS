### Current and clean Windows-player build, execution, and package proof - T4

For each project/output pair below, the same direct editor and player process
arguments were used with explicit process handles and timeouts:

```powershell
$cases = @(
    @(
        (Resolve-Path -LiteralPath 'unity/ReactorGame').Path,
        (Join-Path $root 't4-current-player'),
        't4-current'
    ),
    @(
        (Join-Path $root 'baseline\unity\ReactorGame'),
        (Join-Path $root 't4-clean-player'),
        't4-clean'
    )
)
$editor = Join-Path $env:LOCALAPPDATA `
    'Unity\Editors\6000.3.21f1\Editor\Unity.exe'
foreach ($case in $cases) {
    $project = $case[0]
    $output = $case[1]
    $label = $case[2]
    New-Item -ItemType Directory -Path $output | Out-Null
    $player = Join-Path $output 'ReactorGame.exe'
    $buildLog = Join-Path $root "$label-player-build.log"
    $buildArguments = @(
        '-batchmode', '-nographics', '-quit', '-accept-apiupdate',
        '-projectPath', "`"$project`"",
        '-buildWindows64Player', "`"$player`"",
        '-logFile', "`"$buildLog`""
    )
    $start = New-Object Diagnostics.ProcessStartInfo
    $start.FileName = $editor
    $start.Arguments = $buildArguments -join ' '
    $start.UseShellExecute = $false
    $start.CreateNoWindow = $true
    $process = [Diagnostics.Process]::Start($start)
    try {
        if (-not $process.WaitForExit(120000)) {
            $process.Kill()
            throw "$label player build exceeded 120 seconds."
        }
        if ($process.ExitCode -ne 0) {
            throw "$label player build exited $($process.ExitCode)."
        }
    }
    finally { $process.Dispose() }

    $token = [Guid]::NewGuid().ToString('N')
    $marker = Join-Path $root "$label-player.success"
    $runLog = Join-Path $root "$label-player.log"
    $runArguments = @(
        '-batchmode', '-nographics', '-logFile', "`"$runLog`"",
        '-canduSerializationSmokeMarkerPath', "`"$marker`"",
        '-canduSerializationSmokeMarkerToken', $token
    )
    $runStart = New-Object Diagnostics.ProcessStartInfo
    $runStart.FileName = $player
    $runStart.Arguments = $runArguments -join ' '
    $runStart.UseShellExecute = $false
    $runStart.CreateNoWindow = $true
    $run = [Diagnostics.Process]::Start($runStart)
    try {
        if (-not $run.WaitForExit(30000)) {
            $run.Kill()
            throw "$label player smoke exceeded 30 seconds."
        }
        if ($run.ExitCode -ne 0) {
            throw "$label player smoke exited $($run.ExitCode)."
        }
    }
    finally { $run.Dispose() }

    $expected = $token + "`n" + '{"probe":"reactor-sim","value":42}'
    if ((Get-Content -Raw -LiteralPath $marker) -cne $expected) {
        throw "$label marker mismatch."
    }

    $serializer = Join-Path $output `
        'ReactorGame_Data/Managed/Newtonsoft.Json.dll'
    $packageRoot = @(Get-ChildItem -LiteralPath `
            (Join-Path $project 'Library/PackageCache') -Directory |
        Where-Object { $_.Name -like 'com.unity.nuget.newtonsoft-json@*' })
    $package = Get-Content -Raw -LiteralPath `
        (Join-Path $packageRoot[0].FullName 'package.json') | ConvertFrom-Json
    $aotSerializer = Join-Path $packageRoot[0].FullName `
        'Runtime/AOT/Newtonsoft.Json.dll'
    if ((Get-Item -LiteralPath $serializer).VersionInfo.FileVersion -ne
            '13.0.2' -or
        [string] $package.version -ne '3.2.2' -or
        (Get-FileHash -LiteralPath $serializer -Algorithm SHA256).Hash -ne
            (Get-FileHash -LiteralPath $aotSerializer -Algorithm SHA256).Hash) {
        throw "$label serializer package evidence mismatch."
    }
}
```

Result: both editor builds and both player processes exited 0. Each marker
